--- issue #1: Tasks should carry their creation time ---
`createTask(title)` returns the task with a `createdAt` field set to the moment it was created. Existing fields (`id`, `title`, `done`) are unchanged. `completeTask` keeps `createdAt` as it was. The tests cover the new field.

--- src/tasks.js ---
let nextId = 1;
export function createTask(title) {
  if (typeof title !== "string" || !title.trim()) throw new TypeError("title is required");
  return { id: nextId++, title: title.trim(), done: false };
}
export function completeTask(task) {
  return { ...task, done: true };
}
